# Build stage
FROM mcr.microsoft.com/dotnet/sdk:8.0 AS build
WORKDIR /src

# Copy csproj files and restore
COPY ["src/Pars.API/Pars.API.csproj", "src/Pars.API/"]
COPY ["src/Pars.Infrastructure/Pars.Infrastructure.csproj", "src/Pars.Infrastructure/"]
COPY ["src/Pars.Application/Pars.Application.csproj", "src/Pars.Application/"]
COPY ["src/Pars.Domain/Pars.Domain.csproj", "src/Pars.Domain/"]
RUN dotnet restore "src/Pars.API/Pars.API.csproj"

# Copy everything and build
COPY . .
WORKDIR "/src/src/Pars.API"
RUN dotnet build "Pars.API.csproj" -c Release -o /app/build

# Publish stage
FROM build AS publish
RUN dotnet publish "Pars.API.csproj" -c Release -o /app/publish

# Runtime stage
FROM mcr.microsoft.com/dotnet/aspnet:8.0 AS final
WORKDIR /app
EXPOSE 80
EXPOSE 443
COPY --from=publish /app/publish .
ENTRYPOINT ["dotnet", "Pars.API.dll"]
