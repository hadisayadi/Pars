@code {
    [Parameter] public PersonalFilterRequest Model { get; set; } = new();
    [Parameter] public EventCallback<PersonalFilterRequest> OnFilterChanged { get; set; }

    private List<string> Companies = new();
    private List<UnitOption> UnitOptions = new();
    private System.Timers.Timer? _searchTimer;

    protected override async Task OnInitializedAsync()
    {
        // در پروژه واقعی، اینها از API خوانده می‌شوند
        Companies = new List<string> { "شرکت نفت", "شرکت گاز", "شرکت پخش" };
        UnitOptions = new List<UnitOption>
        {
            new() { Name = "اداری", Count = 45 },
            new() { Name = "فنی", Count = 120 },
            new() { Name = "مالی", Count = 30 },
        };
    }

    private void OnSearchInput()
    {
        // Debounce: ۳۰۰ میلی‌ثانیه صبر کن بعد جستجو کن
        _searchTimer?.Stop();
        _searchTimer = new System.Timers.Timer(300);
        _searchTimer.Elapsed += async (s, e) =>
        {
            _searchTimer!.Stop();
            await InvokeAsync(Apply);
        };
        _searchTimer.Start();
    }

    private void ToggleUnit(UnitOption unit, ChangeEventArgs e)
    {
        unit.IsSelected = (bool)(e.Value ?? false);
        Model.UnitCodes = UnitOptions.Where(u => u.IsSelected).Select(u => u.Name).ToList();
        Apply();
    }

    private void Apply()
    {
        Model.Page = 1; // بازگشت به صفحه اول
        OnFilterChanged.InvokeAsync(Model);
    }

    private void ResetFilters()
    {
        Model = new PersonalFilterRequest();
        UnitOptions.ForEach(u => u.IsSelected = false);
        OnFilterChanged.InvokeAsync(Model);
    }

    public class UnitOption
    {
        public string Name { get; set; } = "";
        public int Count { get; set; }
        public bool IsSelected { get; set; }
    }
}
