@page "/login"
@inject AuthService Auth
@inject NavigationManager Nav

<div class="login-container">
    <div class="login-card">
        <h1>🏢 سیستم مدیریت Pars</h1>
        <p class="subtitle">ورود به حساب کاربری</p>

        @if (!string.IsNullOrEmpty(_error))
        {
            <div class="alert alert-danger">@_error</div>
        }

        <div class="form-group">
            <label>نام کاربری</label>
            <input type="text" @bind="_username" placeholder="نام کاربری خود را وارد کنید" />
        </div>

        <div class="form-group">
            <label>رمز عبور</label>
            <input type="password" @bind="_password" placeholder="رمز عبور" />
        </div>

        <button class="btn-primary" @onclick="HandleLogin" disabled="@_loading">
            @(_loading ? "در حال ورود..." : "ورود")
        </button>
    </div>
</div>

@code {
    private string _username = "";
    private string _password = "";
    private string? _error;
    private bool _loading;

    protected override async Task OnInitializedAsync()
    {
        if (await Auth.IsAuthenticatedAsync())
            Nav.NavigateTo("/");
    }

    private async Task HandleLogin()
    {
        _loading = true;
        _error = null;

        try
        {
            var success = await Auth.LoginAsync(_username, _password);
            if (success)
                Nav.NavigateTo("/");
            else
                _error = "نام کاربری یا رمز عبور اشتباه است";
        }
        catch (Exception ex)
        {
            _error = $"خطا: {ex.Message}";
        }
        finally
        {
            _loading = false;
        }
    }
}
